from datetime import date

from models.task import create_task, is_overdue


def test_task_with_past_due_date_and_not_completed_is_overdue():
    t = create_task("pay rent", "2020-01-01")
    assert is_overdue(t) is True


def test_task_with_past_due_date_but_completed_is_not_overdue():
    t = create_task("pay rent", "2020-01-01")
    t["completed"] = True
    assert is_overdue(t) is False


def test_task_with_no_due_date_is_not_overdue_and_sorts_last():
    t = create_task("someday task")
    assert is_overdue(t) is False


def test_task_due_exactly_today_is_not_yet_overdue():
    today_str = date.today().isoformat()
    t = create_task("today task", today_str)
    assert is_overdue(t) is False

## PR description

**Title:** Add due dates and overdue flagging to tasks

**What changed:** `dueDate` field added to the task model; `isOverdue` computed field added to
API responses; `?sort=dueDate` supported on `GET /tasks`.

**What was tested:** overdue logic (past/today/none/completed cases), sort ordering including
tasks with no due date.

**Known limitations:** due dates cannot be edited after creation; overdue check uses server
timezone only; no reminder/notification.
