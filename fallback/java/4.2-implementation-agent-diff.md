```java
final class Task {
    final int id;
    final String title;
    final String dueDate; // nullable, ISO 8601 (e.g. "2026-07-20")
    boolean completed;

    Task(int id, String title, String dueDate) {
        this.id = id;
        this.title = title;
        this.dueDate = dueDate;
        this.completed = false;
    }
}
```

## `TaskStore.java`

```java
static synchronized Task createTask(String title, String dueDate) {
    Task task = new Task(nextId++, title, dueDate);
    tasks.add(task);
    return task;
}

static boolean isOverdue(Task task) {
    if (task.dueDate == null || task.completed) return false;
    String today = java.time.LocalDate.now().toString();
    return task.dueDate.compareTo(today) < 0;
}
```

## `TaskHandler.java`

```java
private void handleCreate(HttpExchange exchange) throws IOException {
    Map<String, Object> body = Json.parseObject(readBody(exchange));
    Task task = TaskStore.createTask((String) body.get("title"), (String) body.get("dueDate"));
    sendJson(exchange, 201, withOverdueFlag(task));
}

private void handleList(HttpExchange exchange, String query) throws IOException {
    List<Task> tasks = TaskStore.getTasks();
    if ("dueDate".equals(parseSortParam(query))) {
        tasks.sort((a, b) -> {
            if (a.dueDate == null) return 1;
            if (b.dueDate == null) return -1;
            return a.dueDate.compareTo(b.dueDate);
        });
    }
    // ... serialize each via withOverdueFlag(task)
}

private String withOverdueFlag(Task task) {
    return String.format(
        "{\"id\":%d,\"title\":\"%s\",\"completed\":%b,\"dueDate\":%s,\"isOverdue\":%b}",
        task.id, Json.escape(task.title), task.completed,
        task.dueDate == null ? "null" : "\"" + task.dueDate + "\"",
        TaskStore.isOverdue(task));
}
```
