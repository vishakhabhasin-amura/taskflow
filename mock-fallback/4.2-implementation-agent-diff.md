## `models/task.js`

```js
function createTask(title, dueDate = null) {
  return {
    id: nextId++,
    title,
    completed: false,
    dueDate,
  };
}

function isOverdue(task) {
  if (!task.dueDate || task.completed) return false;
  const today = new Date().toISOString().slice(0, 10);
  return task.dueDate < today;
}
```

## `routes/tasks.js`

```js
router.post('/tasks', (req, res) => {
  const { title, dueDate } = req.body;
  const task = createTask(title, dueDate);
  tasks.push(task);
  res.status(201).json(withOverdueFlag(task));
});

router.get('/tasks', (req, res) => {
  let result = tasks.map(withOverdueFlag);
  if (req.query.sort === 'dueDate') {
    result = result.sort((a, b) => {
      if (!a.dueDate) return 1;
      if (!b.dueDate) return -1;
      return a.dueDate.localeCompare(b.dueDate);
    });
  }
  res.json(result);
});

function withOverdueFlag(task) {
  return { ...task, isOverdue: isOverdue(task) };
}
```
